NEW FILES:

1. include/aurora/cubemap.hpp + src/cubemap.cpp — Cubemap texture class:
   - Constructor takes std::array<std::string, 6> face_paths in this order:
       0: +X (right)
       1: -X (left)
       2: +Y (top)
       3: -Y (bottom)
       4: +Z (front)
       5: -Z (back)
     This matches OpenGL's GL_TEXTURE_CUBE_MAP_POSITIVE_X..._NEGATIVE_Z layout.
   - For each face: stbi_load (NO vertical flip for cubemaps - that's a 2D texture convention; cubemaps use as-is), glTexImage2D(GL_TEXTURE_CUBE_MAP_POSITIVE_X + i, ...).
   - Texture parameters:
       glTexParameteri(GL_TEXTURE_CUBE_MAP, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
       glTexParameteri(GL_TEXTURE_CUBE_MAP, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
       glTexParameteri(GL_TEXTURE_CUBE_MAP, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
       glTexParameteri(GL_TEXTURE_CUBE_MAP, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
       glTexParameteri(GL_TEXTURE_CUBE_MAP, GL_TEXTURE_WRAP_R, GL_CLAMP_TO_EDGE);
   - Log success or failure per-face via spdlog.
   - bind(unsigned int unit) — glActiveTexture + glBindTexture(GL_TEXTURE_CUBE_MAP, id).
   - id() and unit() accessors.
   - RAII destructor; movable, not copyable.

2. include/aurora/skybox.hpp + src/skybox.cpp — Skybox class:
   - Constructor takes std::shared_ptr<Cubemap>.
   - Owns: VAO, VBO for a 1x1x1 unit cube (positions only, 36 vertices, no indices for simplicity).
   - Owns its own Shader (built from shaders/skybox.vert + shaders/skybox.frag — see below).
   - draw(const glm::mat4& view, const glm::mat4& projection):
       glDepthFunc(GL_LEQUAL);          // allow depth = 1.0 to pass
       shader.bind();
       glm::mat4 view_no_translate = glm::mat4(glm::mat3(view));   // strip translation
       shader.set_uniform("u_view", view_no_translate);
       shader.set_uniform("u_projection", projection);
       cubemap->bind(0);
       shader.set_uniform("u_skybox", 0);
       glBindVertexArray(vao);
       glDrawArrays(GL_TRIANGLES, 0, 36);
       glDepthFunc(GL_LESS);            // restore
   - RAII destructor.

NEW SHADERS:

shaders/skybox.vert:
   #version 460 core
   layout(location = 0) in vec3 a_position;
   out vec3 v_local_dir;

   uniform mat4 u_view;
   uniform mat4 u_projection;

   void main() {
       v_local_dir = a_position;
       vec4 pos = u_projection * u_view * vec4(a_position, 1.0);
       gl_Position = pos.xyww;   // force z = w => depth = 1.0 in NDC (max depth)
   }

shaders/skybox.frag:
   #version 460 core
   in vec3 v_local_dir;
   out vec4 frag_color;

   uniform samplerCube u_skybox;

   void main() {
       frag_color = texture(u_skybox, v_local_dir);
   }
